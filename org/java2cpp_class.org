#+begin_src java :results output :cmdline < input
  // Main.java - Read time from user and compute free-fall distance
  // Formula: d = (1/2) * g * t^2, g = 9.81 m/s^2

  import java.util.Scanner;

  public class Main {
      public static void main(String[] args) {

          // declare constants and variables
          final double G = 9.81;
          final double HALF = 0.5;
          double time;
          double distance;

          // create scanner for user input
          Scanner input = new Scanner(System.in);

          // prompt user
          System.out.println("Enter time in seconds:");
          time = input.nextDouble();

          // compute distance
          distance = HALF * G * time * time;

          // print result
          System.out.printf(
                            "After %.0f seconds, the object has fallen %.3f meters.%n",
                            time, distance        );
      }
  }
